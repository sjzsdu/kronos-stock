{# 通用按钮组件 参数: - btn_id: 按钮ID - btn_type: 按钮类型 (button, submit, reset) - btn_text: 按钮文本 - btn_style: 按钮样式 (primary, secondary, success, warning, danger) - btn_size: 按钮大小 (small, medium, large) - icon: FontAwesome图标类名 - icon_position: 图标位置 (left, right) - disabled: 是否禁用 (boolean) - loading: 是否显示加载状态 (boolean) - onclick: 点击事件处理 - htmx_attributes: HTMX属性字典 {'post': '/api/url', 'target': '#id', ...} - additional_classes: 额外的CSS类 #}
